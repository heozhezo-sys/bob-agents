- [ ] **Step 1: Write the failing check**

Create `pro/scripts/css-check.mjs`:

```js
#!/usr/bin/env node
/**
 * Guards against a stylesheet that silently loses every rule after a break.
 * An unclosed rule makes the browser drop the rest of the file, which no
 * type check, API smoke or render smoke can see.
 */
import { readFile } from "node:fs/promises";
import path from "node:path";

const ROOT = path.resolve(import.meta.dirname, "..");
const STYLESHEETS = ["client/src/styles.css"];

let failures = 0;

for (const relative of STYLESHEETS) {
  const source = await readFile(path.join(ROOT, relative), "utf8");
  const stripped = source
    .replace(/\/\*[\s\S]*?\*\//g, "")
    .replace(/"(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*'/g, '""');

  let depth = 0;
  let line = 1;
  let extraAt = 0;
  for (const char of stripped) {
    if (char === "\n") line += 1;
    else if (char === "{") depth += 1;
    else if (char === "}") {
      depth -= 1;
      if (depth < 0) {
        extraAt = line;
        break;
      }
    }
  }

  if (depth !== 0 || extraAt > 0) {
    failures += 1;
    const detail = extraAt > 0 ? `unexpected } near line ${extraAt}` : `${depth} unclosed {`;
    console.error(`  FAIL  ${relative} has unbalanced braces — ${detail}`);
  } else {
    console.log(`  PASS  ${relative} has balanced braces`);
  }
}

if (failures > 0) {
  console.error(`${failures} stylesheet check(s) failed.`);
  process.exitCode = 1;
} else {
  console.log("All stylesheet checks passed.");
}
```

### Task 6: Purge the leftover smoke data

**Files:**
- Create: `pro/scripts/purge-smoke.mjs`

**Interfaces:**
- Consumes: `GET /api/projects` and `DELETE /api/projects/:id` (the delete removes asset files, and after Task 4 the render directory too).
- Produces: `npm run purge:smoke` — deletes every project whose name matches a smoke-test pattern, then removes render directories under `data/renders/` whose project no longer exists.

- [ ] **Step 1: Write the purge script**

Create `pro/scripts/purge-smoke.mjs`:

```js
#!/usr/bin/env node
/**
 * Deletes leftover smoke-test projects and any render directory whose project
 * no longer exists. The user's own projects are never touched: only names that
 * match a smoke-test pattern are removed.
 */
import { readdir, rm } from "node:fs/promises";
import path from "node:path";

const BASE = process.env.BASE_URL ?? "http://127.0.0.1:4320";
const PASSWORD = process.env.APP_PASSWORD ?? "cutroom-dev";
const ROOT = path.resolve(import.meta.dirname, "..");
const RENDERS = path.join(ROOT, "data", "renders");
const SMOKE_NAMES = [/^UI upload /, /^Real footage smoke /, /^Render smoke$/, /^Leak probe$/];

const loginResponse = await fetch(`${BASE}/api/auth/login`, {
  method: "POST",
  headers: { "content-type": "application/json" },
  body: JSON.stringify({ password: PASSWORD }),
});
if (!loginResponse.ok) throw new Error(`Login failed: ${loginResponse.status}`);
const cookie = (loginResponse.headers.getSetCookie?.() ?? [loginResponse.headers.get("set-cookie") ?? ""])
  .filter(Boolean)
  .map((value) => value.split(";")[0])
  .join("; ");
const csrf = (await loginResponse.json()).csrfToken;

const listResponse = await fetch(`${BASE}/api/projects`, { headers: { cookie } });
const { projects } = await listResponse.json();

let removed = 0;
for (const project of projects) {
  if (!SMOKE_NAMES.some((pattern) => pattern.test(project.name))) continue;
  const response = await fetch(`${BASE}/api/projects/${project.id}`, {
    method: "DELETE",
    headers: { cookie, "x-csrf-token": csrf },
  });
  console.log(`  ${response.status === 200 ? "removed" : "FAILED"}  ${project.name}`);
  removed += 1;
}
console.log(`\nRemoved ${removed} smoke project(s).`);

const live = new Set(projects.map((project) => project.id));
try {
  for (const entry of await readdir(RENDERS, { withFileTypes: true })) {
    if (!entry.isDirectory() || live.has(entry.name)) continue;
    await rm(path.join(RENDERS, entry.name), { recursive: true, force: true });
    console.log(`  removed  orphaned render directory ${entry.name}`);
  }
} catch (error) {
  if (error.code !== "ENOENT") throw error;
}
```
